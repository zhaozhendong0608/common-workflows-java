## 🔧 分层架构

### 三层架构

```python
# 1. 模型层 (Model) - models.py
from django.db import models

class User(models.Model):
    """用户模型"""
    username = models.CharField(max_length=50, unique=True)
    email = models.EmailField(unique=True)
    created_at = models.DateTimeField(auto_now_add=True)
    
    class Meta:
        db_table = 'users'
        ordering = ['-created_at']

# 2. 业务逻辑层 (Service) - services.py
class UserService:
    """用户业务逻辑"""
    
    @staticmethod
    def create_user(username: str, email: str) -> User:
        """创建用户"""
        # 验证逻辑
        if User.objects.filter(email=email).exists():
            raise ValueError("Email already exists")
        
        # 创建用户
        user = User.objects.create(
            username=username,
            email=email
        )
        return user
    
    @staticmethod
    def get_user_by_id(user_id: int) -> User:
        """获取用户"""
        try:
            return User.objects.get(id=user_id)
        except User.DoesNotExist:
            raise ValueError(f"User {user_id} not found")

# 3. 视图层 (View/Controller) - views.py
from rest_framework.views import APIView
from rest_framework.response import Response

class UserView(APIView):
    """用户视图"""
    
    def post(self, request):
        """创建用户"""
        try:
            user = UserService.create_user(
                username=request.data['username'],
                email=request.data['email']
            )
            return Response({'id': user.id}, status=201)
        except ValueError as e:
            return Response({'error': str(e)}, status=400)
```

## 🗃️ 数据库规范

### ORM 最佳实践

```python
# ✅ 正确 - 使用 select_related 减少查询
users = User.objects.select_related('profile').all()

# ❌ 错误 - N+1 查询问题
users = User.objects.all()
for user in users:
    print(user.profile.bio)  # 每次循环都查询一次

# ✅ 正确 - 使用 prefetch_related
users = User.objects.prefetch_related('orders').all()

# ✅ 正确 - 批量创建
User.objects.bulk_create([
    User(username='user1', email='user1@example.com'),
    User(username='user2', email='user2@example.com'),
])

# ❌ 错误 - 循环创建
for i in range(100):
    User.objects.create(username=f'user{i}')  # 100次数据库写入
```

```python
# ✅ 正确 - 使用 Django 内置
from django.contrib.auth.hashers import make_password, check_password

hashed = make_password('my_password')
is_valid = check_password('my_password', hashed)

```

### SQL 注入防护

```python
# ✅ 正确 - 参数化查询
User.objects.filter(username=user_input)

# ✅ 正确 - 原始查询使用参数
from django.db import connection
cursor = connection.cursor()
cursor.execute("SELECT * FROM users WHERE id = %s", [user_id])

# ❌ 危险 - 字符串拼接
cursor.execute(f"SELECT * FROM users WHERE id = {user_id}")  # SQL 注入风险
```

## 🧪 测试规范

### 单元测试

```python
# tests/test_services.py
import pytest
from django.test import TestCase
from myapp.services import UserService

class TestUserService(TestCase):
    """用户服务测试"""
    
    def setUp(self):
        """测试准备"""
        self.test_email = 'test@example.com'
    
    def test_create_user_success(self):
        """测试创建用户成功"""
        user = UserService.create_user(
            username='testuser',
            email=self.test_email
        )
        self.assertIsNotNone(user.id)
        self.assertEqual(user.email, self.test_email)
    
    def test_create_user_duplicate_email(self):
        """测试重复邮箱"""
        UserService.create_user('user1', self.test_email)
        
        with self.assertRaises(ValueError):
            UserService.create_user('user2', self.test_email)

# 运行测试
python manage.py test
pytest
```

## 🚀 异步编程

### asyncio 规范

```python
# ✅ 正确 - async/await
import asyncio

async def fetch_user(user_id: int):
    """异步获取用户"""
    # 模拟异步 I/O
    await asyncio.sleep(1)
    return {"id": user_id, "name": "User"}

async def main():
    """主函数"""
    # 并发执行
    users = await asyncio.gather(
        fetch_user(1),
        fetch_user(2),
        fetch_user(3)
    )
    return users

# 运行
asyncio.run(main())
```
