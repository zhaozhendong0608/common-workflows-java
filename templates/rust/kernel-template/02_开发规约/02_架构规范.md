## 🔧 分层架构

### 完整示例

```rust
// 1. Domain 层 (src/domain/user.rs)
use serde::{Deserialize, Serialize};
use chrono::{DateTime, Utc};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct User {
    pub id: i64,
    pub username: String,
    pub email: String,
    pub created_at: DateTime<Utc>,
}

// Repository trait（依赖倒置）
#[async_trait::async_trait]
pub trait UserRepository: Send + Sync {
    async fn create(&self, user: &User) -> Result<User>;
    async fn find_by_id(&self, id: i64) -> Result<Option<User>>;
    async fn find_by_email(&self, email: &str) -> Result<Option<User>>;
}

// 2. Infrastructure 层 (src/infrastructure/repository/user.rs)
use sqlx::{PgPool, FromRow};
use async_trait::async_trait;

pub struct PostgresUserRepository {
    pool: PgPool,
}

impl PostgresUserRepository {
    pub fn new(pool: PgPool) -> Self {
        Self { pool }
    }
}

#[async_trait]
impl UserRepository for PostgresUserRepository {
    async fn create(&self, user: &User) -> Result<User> {
        let user = sqlx::query_as::<_, User>(
            "INSERT INTO users (username, email) VALUES ($1, $2) RETURNING *"
        )
        .bind(&user.username)
        .bind(&user.email)
        .fetch_one(&self.pool)
        .await
        .map_err(|e| Error::Database(e.to_string()))?;
        
        Ok(user)
    }
    
    async fn find_by_id(&self, id: i64) -> Result<Option<User>> {
        let user = sqlx::query_as::<_, User>(
            "SELECT * FROM users WHERE id = $1"
        )
        .bind(id)
        .fetch_optional(&self.pool)
        .await
        .map_err(|e| Error::Database(e.to_string()))?;
        
        Ok(user)
    }
}

// 3. Application 层 (src/application/user_service.rs)
use std::sync::Arc;

pub struct UserService {
    repository: Arc<dyn UserRepository>,
}

impl UserService {
    pub fn new(repository: Arc<dyn UserRepository>) -> Self {
        Self { repository }
    }
    
    pub async fn create_user(
        &self,
        username: String,
        email: String,
    ) -> Result<User> {
        // 业务验证
        if username.is_empty() {
            return Err(Error::Validation("Username required".into()));
        }
        
        // 检查重复
        if let Some(_) = self.repository.find_by_email(&email).await? {
            return Err(Error::Conflict("Email already exists".into()));
        }
        
        let user = User {
            id: 0,  // 由数据库生成
            username,
            email,
            created_at: Utc::now(),
        };
        
        self.repository.create(&user).await
    }
}

// 4. API 层 (src/api/handlers.rs) - Axum 示例
use axum::{
    extract::{State, Path},
    Json,
    response::IntoResponse,
    http::StatusCode,
};

#[derive(Deserialize)]
pub struct CreateUserRequest {
    username: String,
    email: String,
}

pub async fn create_user(
    State(service): State<Arc<UserService>>,
    Json(req): Json<CreateUserRequest>,
) -> Result<impl IntoResponse> {
    let user = service
        .create_user(req.username, req.email)
        .await?;
    
    Ok((StatusCode::CREATED, Json(user)))
}

pub async fn get_user(
    State(service): State<Arc<UserService>>,
    Path(id): Path<i64>,
) -> Result<impl IntoResponse> {
    let user = service
        .repository
        .find_by_id(id)
        .await?
        .ok_or(Error::NotFound)?;
    
    Ok(Json(user))
}
```

```rust
use sqlx::{PgPool, postgres::PgPoolOptions};

// ✅ 正确 - 连接池配置
pub async fn create_pool(database_url: &str) -> Result<PgPool> {
    PgPoolOptions::new()
        .max_connections(20)
        .min_connections(5)
        .acquire_timeout(std::time::Duration::from_secs(5))
        .connect(database_url)
        .await
        .map_err(|e| Error::Database(e.to_string()))
}

// ✅ 正确 - 编译期检查查询
let user = sqlx::query_as!(
    User,
    "SELECT id, username, email, created_at FROM users WHERE id = $1",
    id
)
.fetch_one(&pool)
.await?;

// ✅ 正确 - 事务
let mut tx = pool.begin().await?;

sqlx::query("INSERT INTO users ...")
    .execute(&mut *tx)
    .await?;

sqlx::query("INSERT INTO profiles ...")
    .execute(&mut *tx)
    .await?;

tx.commit().await?;
```

### 密码加密

```rust
use argon2::{
    password_hash::{PasswordHash, PasswordHasher, PasswordVerifier, SaltString},
    Argon2,
};
use rand::rngs::OsRng;

// ✅ 正确 - 加密密码
pub fn hash_password(password: &str) -> Result<String> {
    let salt = SaltString::generate(&mut OsRng);
    let argon2 = Argon2::default();
    
    let hash = argon2
        .hash_password(password.as_bytes(), &salt)
        .map_err(|e| Error::Internal(e.to_string()))?
        .to_string();
    
    Ok(hash)
}

// ✅ 正确 - 验证密码
pub fn verify_password(password: &str, hash: &str) -> Result<bool> {
    let parsed_hash = PasswordHash::new(hash)
        .map_err(|e| Error::Internal(e.to_string()))?;
    
    Ok(Argon2::default()
        .verify_password(password.as_bytes(), &parsed_hash)
        .is_ok())
}
```

### JWT 认证

```rust
use jsonwebtoken::{encode, decode, Header, Validation, EncodingKey, DecodingKey};
use serde::{Deserialize, Serialize};

#[derive(Debug, Serialize, Deserialize)]
pub struct Claims {
    pub sub: i64,  // user_id
    pub exp: i64,  // expiration
}

pub fn generate_token(user_id: i64, secret: &str) -> Result<String> {
    let claims = Claims {
        sub: user_id,
        exp: (chrono::Utc::now() + chrono::Duration::hours(24)).timestamp(),
    };
    
    encode(
        &Header::default(),
        &claims,
        &EncodingKey::from_secret(secret.as_bytes()),
    )
    .map_err(|e| Error::Internal(e.to_string()))
}

pub fn validate_token(token: &str, secret: &str) -> Result<Claims> {
    decode::<Claims>(
        token,
        &DecodingKey::from_secret(secret.as_bytes()),
        &Validation::default(),
    )
    .map(|data| data.claims)
    .map_err(|e| Error::Unauthorized(e.to_string()))
}
```
