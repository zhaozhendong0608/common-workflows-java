// tests/integration/dept.spec.js
describe('部门管理 - 业务流程测试', () => {

 test('正常流程: 创建 → 查询 → 修改 → 删除', async () => {
 // 1. 创建部门
 const createResp = await api.post('/api/dept', {
 deptName: '研发部',
 parentId: 0
 });
 expect(createResp.status).toBe(200);
 const deptId = createResp.data.id;

 // 2. 查询部门
 const getResp = await api.get(\`/api/dept/\${deptId}\`);
 expect(getResp.data.deptName).toBe('研发部');

 // 3. 修改部门
 const updateResp = await api.put(\`/api/dept/\${deptId}\`, {
 deptName: '技术研发部'
 });
 expect(updateResp.status).toBe(200);

 // 4. 删除部门
 const deleteResp = await api.delete(\`/api/dept/\${deptId}\`);
 expect(deleteResp.status).toBe(200);
 });

 test('异常场景: 删除有子部门的父部门', async () => {
 // 1. 创建父部门
 const parent = await api.post('/api/dept', {
 deptName: '研发部'
 });

 // 2. 创建子部门
 await api.post('/api/dept', {
 deptName: '后端组',
 parentId: parent.data.id
 });

 // 3. 尝试删除父部门
 const deleteResp = await api.delete(\`/api/dept/\${parent.data.id}\`);

 // 4. 期望失败
 expect(deleteResp.status).toBe(400);
 expect(deleteResp.data.message).toContain('存在子部门');
 });

 test('业务规则: 部门名称不能重复', async () => {
 // 1. 创建第一个部门
 await api.post('/api/dept', {
 deptName: '研发部'
 });

 // 2. 尝试创建同名部门
 const resp = await api.post('/api/dept', {
 deptName: '研发部'
 });

 // 3. 期望失败
 expect(resp.status).toBe(400);
 expect(resp.data.message).toContain('部门名称已存在');
 });
});
\`\`\`

// tests/e2e/dept-manage.spec.js
import { test, expect } from '@playwright/test';

test.describe('部门管理 - E2E 测试', () => {

 test('完整业务流程', async ({ page }) => {
 // 1. 登录
 await page.goto('http://localhost:8080/login');

 await page.fill('#password', '123456');

 await expect(page).toHaveURL('/dashboard');

 await page.click('text=部门管理');
 await expect(page).toHaveURL('/dept-manage');

 // 3. 创建部门
 await page.click('button:has-text("新增部门")');
 await page.fill('#deptName', '测试部门');
 await page.click('button:has-text("确定")');

 // 4. 验证创建成功
 await expect(page.locator('text=测试部门')).toBeVisible();

 // 5. 编辑部门
 await page.click('text=测试部门 >> xpath=../.. >> button:has-text("编辑")');
 await page.fill('#deptName', '测试部门-修改');
 await page.click('button:has-text("确定")');

 // 6. 验证修改成功
 await expect(page.locator('text=测试部门-修改')).toBeVisible();

 // 7. 删除部门
 await page.click('text=测试部门-修改 >> xpath=../.. >> button:has-text("删除")');
 await page.click('button:has-text("确认")');

 // 8. 验证删除成功
 await expect(page.locator('text=测试部门-修改')).not.toBeVisible();
 });

 test('异常场景: 删除有子部门的父部门', async ({ page }) => {
 // ... 省略登录步骤

 // 1. 创建父部门
 await page.click('button:has-text("新增部门")');
 await page.fill('#deptName', '父部门');
 await page.click('button:has-text("确定")');

 // 2. 创建子部门
 await page.click('text=父部门 >> xpath=../.. >> button:has-text("新增子部门")');
 await page.fill('#deptName', '子部门');
 await page.click('button:has-text("确定")');

 // 3. 尝试删除父部门
 await page.click('text=父部门 >> xpath=../.. >> button:has-text("删除")');
 await page.click('button:has-text("确认")');

 // 4. 验证错误提示
 await expect(page.locator('.el-message--error')).toContainText('存在子部门');
 });
});

### D.1 接口性能测试

**使用 k6**:

\`\`\`javascript
// tests/performance/dept-api.js
import http from 'k6/http';
import { check, sleep } from 'k6';

export let options = {
 stages: [
 { duration: '1m', target: 10 }, // 1分钟内达到 10 并发
 { duration: '3m', target: 50 }, // 3分钟内达到 50 并发
 { duration: '1m', target: 0 }, // 1分钟内降至 0
 ],
 thresholds: {
 http_req_duration: ['p(95)<500'], // 95% 请求响应时间 < 500ms
 },
};

export default function () {
 // 查询部门列表
 let res = http.get('http://localhost:8080/api/dept/list');

 check(res, {
 'status is 200': (r) => r.status === 200,
 'response time < 500ms': (r) => r.timings.duration < 500,
 });

 sleep(1);
}

## 使用技巧

### 测试优先级

**P0（必须测试）**:
- ✅ 核心业务流程（CRUD）
- ✅ 异常场景（删除有子部门）
- ✅ 数据一致性

**P1（建议测试）**:
- ⚠️ UI/UX 体验
- ⚠️ 边界条件
- ⚠️ 性能压测

**P2（可选测试）**:
- ⏸️ 浏览器兼容性（非 Chrome）
- ⏸️ 移动端适配

### 自动化 vs 人工

| 测试类型 | 自动化 | 人工 | 建议 |
|---------|--------|------|------|
| API 测试 | ✅ | ❌ | 100% 自动化 |
| E2E 测试 | ✅ | ⚠️ | 80% 自动化 + 20% 人工复查 |
| UI 美观度 | ❌ | ✅ | 100% 人工 |
| 性能测试 | ✅ | ❌ | 100% 自动化 |

## 常见问题

**Q: 业务测试是否必须执行？**
A:
- 新功能开发: ✅ 必须
- Bug 修复: ⚠️ 建议（至少回归测试）
- 重构: ✅ 必须（确保行为不变）

**Q: 人工测试太耗时怎么办？**
A:
